cmake_minimum_required(VERSION 3.24)
project(pryx_page_generator LANGUAGES CXX ASM)
add_compile_options($<$<COMPILE_LANGUAGE:CXX>:-fno-exceptions> $<$<COMPILE_LANGUAGE:CXX>:-fno-rtti>)
set(ASMJIT_STATIC ON CACHE BOOL "" FORCE)
set(ASMJIT_TEST OFF CACHE BOOL "" FORCE)
set(ASMJIT_NO_FOREIGN ON CACHE BOOL "" FORCE)
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
add_subdirectory("${ASMJIT_SOURCE}" asmjit)
add_library(pryx_page_machine STATIC src/generator.cpp)
target_compile_features(pryx_page_machine PRIVATE cxx_std_20)
target_link_libraries(pryx_page_machine PRIVATE asmjit::asmjit)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
  target_sources(pryx_page_machine PRIVATE src/aarch64.S)
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64)$")
  target_sources(pryx_page_machine PRIVATE src/x86_64.S)
else()
  message(FATAL_ERROR "Unsupported generator architecture")
endif()
